        // Init: Word Counter & Clear Fields

                document.getElementById('wordCount').textContent = words;

        // UI Interactions
        function selectTask(element, taskId) {
            // Reset visual states
            document.querySelectorAll('.task-card').forEach(card => {
                card.classList.remove('task-card-active', 'border-blue-500');
                card.classList.add('border-slate-300');
            });
            
            // Highlight selected
            element.classList.add('task-card-active', 'border-blue-500');
            element.classList.remove('border-slate-300');
            
            // Adjust placeholder text dynamically based on the exact task chosen
            const promptArea = document.getElementById('promptInput');
            if(taskId === 'task1_general') {
                promptArea.placeholder = "E.g., You recently stayed in a hotel... Write a letter to the manager expressing your dissatisfaction...";
            } else if (taskId === 'task1_academic') {
                promptArea.placeholder = "E.g., The provided bar chart shows the percentage of households with internet access...";
            } else {
                promptArea.placeholder = "E.g., Some people believe that technology has made our lives more complex. To what extent do you agree or disagree?";
            }
        }

        function switchView(viewId) {
            document.querySelectorAll('.view-section').forEach(view => {
                view.classList.add('hidden');
            });
            
            const targetView = document.getElementById(`view-${viewId}`);
            if(targetView) {
                targetView.classList.remove('hidden');
                if(viewId === 'results') {
                    targetView.classList.add('flex');
                }
            }
            window.scrollTo(0, 0);
        }

        // Helper function to escape HTML to prevent XSS and format newlines
        function formatTextToHTML(text) {
            if (!text) return "";
            // Escape basic HTML characters
            let escaped = text.replace(/&/g, "&amp;")
                              .replace(/</g, "&lt;")
                              .replace(/>/g, "&gt;");
            // Split by double newline to create paragraphs
            let paragraphs = escaped.split(/\n\s*\n/);
            return paragraphs.map(p => `<p class="mb-3">${p.replace(/\n/g, '<br>')}</p>`).join('');
        }

        function startAnalysis() {
            // 1. Get User Data
            const targetScore = document.getElementById('targetScore').value;
            const studentName = document.getElementById('studentName').value || 'Anonymous';
            const essayText = document.getElementById('essayInput').value;
            const promptText = document.getElementById('promptInput').value;

            // 2. Validate
            if(!essayText.trim()) {
                alert("Please enter your response in the text area before submitting.");
                return;
            }

            // 3. Inject User Data into Results Page
            document.getElementById('display-student-name').textContent = studentName;
            document.getElementById('display-target-score').textContent = targetScore;
            document.getElementById('text-target-score').textContent = targetScore;
            document.getElementById('rewrite-target').textContent = targetScore;

            // Inject the EXACT text the user typed
            const resultPromptDiv = document.getElementById('result-prompt');
            resultPromptDiv.innerHTML = formatTextToHTML(promptText) || "<em>No prompt provided.</em>";

            const resultEssayDiv = document.getElementById('result-essay');
            resultEssayDiv.innerHTML = formatTextToHTML(essayText);

            // 4. Show Loading State
            switchView('loading');
            
            const loadingTexts = [
                "Identifying key strengths...",
                "Analyzing grammar and vocabulary...",
                "Evaluating task achievement...",
                "Checking coherence and cohesion...",
                "Generating detailed report..."
            ];
            let textIndex = 0;
            const textElement = document.getElementById('loadingText');
            
            const loadingInterval = setInterval(() => {
                textIndex = (textIndex + 1) % loadingTexts.length;
                textElement.textContent = loadingTexts[textIndex];
            }, 800);

            // 5. Simulate processing delay, then show results
            setTimeout(() => {
                clearInterval(loadingInterval);
                switchView('results');
                
                // Trigger gauge animation
                const circle = document.querySelector('.circular-chart .circle');
                circle.style.animation = 'none';
                circle.offsetHeight; 
                circle.style.animation = null; 

                // Make sure Overview tab is active by default
                switchTab('overview');

            }, 3500);
        }

        function switchTab(tabId) {
            document.querySelectorAll('.tab-content').forEach(content => {
                content.classList.add('hidden');
            });
            document.querySelectorAll('.tab-btn').forEach(btn => {
                btn.classList.remove('tab-active', 'text-slate-800');
                btn.classList.add('text-slate-500');
            });

            document.getElementById(`tab-${tabId}`).classList.remove('hidden');
            
            const activeBtn = document.getElementById(`tab-btn-${tabId}`);
            activeBtn.classList.add('tab-active', 'text-slate-800');
            activeBtn.classList.remove('text-slate-500');
        }

        function toggleAccordion(id) {
            const content = document.getElementById(id);
            const icon = document.getElementById(`icon-${id}`);
            
            if(content.classList.contains('hidden')) {
                content.classList.remove('hidden');
                content.classList.add('block');
                icon.classList.remove('fa-chevron-down');
                icon.classList.add('fa-chevron-up');
                content.parentElement.classList.add('border-l-4', 'border-l-blue-500', 'shadow-sm');
            } else {
                content.classList.remove('block');
                content.classList.add('hidden');
                icon.classList.remove('fa-chevron-up');
                icon.classList.add('fa-chevron-down');
                content.parentElement.classList.remove('border-l-4', 'border-l-blue-500', 'shadow-sm');
            }
        }
